

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

from sklearn.datasets import load_iris
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans, DBSCAN
from sklearn.mixture import GaussianMixture
from sklearn.model_selection import cross_val_predict, StratifiedKFold
from sklearn.tree import DecisionTreeClassifier
from sklearn.dummy import DummyClassifier
from sklearn.metrics import (
    accuracy_score, confusion_matrix, classification_report,
    adjusted_rand_score, roc_curve, auc
)
from sklearn.preprocessing import label_binarize

# Load Iris dataset
iris = load_iris()
X = iris.data
y = iris.target

# Standardize numeric attributes
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# Part A: K-Means clustering
print("K-MEANS CLUSTERING")

sse_values = []

for k in [2, 3, 4, 5]:
    model = KMeans(n_clusters=k, random_state=10, n_init=10)
    clusters = model.fit_predict(X_scaled)

    print("\nK =", k)
    print("SSE:", round(model.inertia_, 3))
    print("Cluster counts:", np.bincount(clusters))
    print("Centroids (standardized):")
    print(np.round(model.cluster_centers_, 3))
    print("Adjusted Rand Index:", round(
        adjusted_rand_score(y, clusters), 3
    ))

    sse_values.append(model.inertia_)

# Elbow graph
plt.figure()
plt.plot([2, 3, 4, 5], sse_values, marker="o")
plt.xlabel("Number of clusters (k)")
plt.ylabel("SSE")
plt.title("K-Means Elbow Method")
plt.grid(True)
plt.show()

# Visualize K-Means clusters using petal measurements
kmeans = KMeans(n_clusters=3, random_state=10, n_init=10)
clusters = kmeans.fit_predict(X_scaled)

plt.figure()
plt.scatter(X[:, 2], X[:, 3], c=clusters)
plt.xlabel("Petal Length")
plt.ylabel("Petal Width")
plt.title("Iris K-Means Clusters (k=3)")
plt.show()

# EM clustering
print("\nEM CLUSTERING")
em = GaussianMixture(n_components=3, random_state=10)
em_clusters = em.fit_predict(X_scaled)
print("Cluster counts:", np.bincount(em_clusters))
print("Adjusted Rand Index:", round(
    adjusted_rand_score(y, em_clusters), 3
))

# DBSCAN clustering
print("\nDBSCAN CLUSTERING")
dbscan = DBSCAN(eps=0.7, min_samples=5)
db_clusters = dbscan.fit_predict(X_scaled)
print("Cluster labels (-1 means noise):", np.unique(db_clusters))
print("Cluster counts (including noise):",
      dict(zip(*np.unique(db_clusters, return_counts=True))))
print("Adjusted Rand Index:", round(
    adjusted_rand_score(y, db_clusters), 3
))

# Part B: Classification using 10-fold cross-validation
print("\nCLASSIFICATION: J48-STYLE TREE AND ZEROR")

cv = StratifiedKFold(n_splits=10, shuffle=True, random_state=10)

classifiers = {
    "J48-style Decision Tree": DecisionTreeClassifier(
        random_state=10
    ),
    "ZeroR baseline": DummyClassifier(strategy="most_frequent")
}

for name, classifier in classifiers.items():
    predictions = cross_val_predict(classifier, X, y, cv=cv)

    print("\nClassifier:", name)
    print("Accuracy:", round(
        accuracy_score(y, predictions) * 100, 2
    ), "%")
    print("Confusion Matrix:")
    print(confusion_matrix(y, predictions))
    print(classification_report(
        y, predictions, target_names=iris.target_names
    ))

# ROC curves using cross-validated probabilities
y_binary = label_binarize(y, classes=[0, 1, 2])

plt.figure()

for name, classifier in classifiers.items():
    probabilities = cross_val_predict(
        classifier, X, y, cv=cv, method="predict_proba"
    )

    # Micro-average ROC for multiclass classification
    fpr, tpr, _ = roc_curve(
        y_binary.ravel(), probabilities.ravel()
    )
    roc_auc = auc(fpr, tpr)

    plt.plot(fpr, tpr, label=f"{name} (AUC={roc_auc:.3f})")

plt.plot([0, 1], [0, 1], linestyle="--")
plt.xlabel("False Positive Rate")
plt.ylabel("True Positive Rate")
plt.title("ROC Curves - Iris Dataset")
plt.legend()
plt.grid(True)
plt.show()

print("\nExperiment completed.")
